   307→   307→def _extract_clean_json(text: str) -> str:
   308→   308→    """
   309→   309→    يستخرج JSON نظيف من نص قد يحتوي على علامات Markdown أو نص إضافي.
   310→   310→    يتعامل مع:
   311→   311→    - ```json ... ```
   312→   312→    - ``` ... ```
   313→   313→    - نص قبل/بعد الـ JSON
   314→   314→    - الأسطر الجديدة الزائدة
   315→   315→    """
   316→   316→    if not text:
   317→   317→        return ""
   318→   318→
   319→   319→    text = text.strip()
   320→   320→
   321→   321→    # إزالة علامات Markdown code block
   322→   322→    if "```" in text:
   323→   323→        # استخراج المحتوى بين علامات ```
   324→   324→        import re as _re
   325→   325→        # pattern يطابق ```json أو ``` في البداية، و ``` في النهاية
   326→   326→        match = _re.search(r'```(?:json)?\s*\n?(.*?)```', text, _re.DOTALL)
   327→   327→        if match:
   328→   328→            text = match.group(1).strip()
   329→   329→        else:
   330→   330→            # لو فيه ``` بس ما اكتمل النمط، نزيلها يدوياً
   331→   331→            text = text.replace("```json", "").replace("```", "").strip()
   332→   332→
   333→   333→    # إزالة أي نص قبل أول { أو [
   334→   334→    first_brace = -1
   335→   335→    for i, char in enumerate(text):
   336→   336→        if char in ('{', '['):
   337→   337→            first_brace = i
   338→   338→            break
   339→   339→
   340→   340→    if first_brace > 0:
   341→   341→        text = text[first_brace:]
   342→   342→
   343→   343→    # إزالة أي نص بعد آخر } أو ]
   344→   344→    last_brace = -1
   345→   345→    for i in range(len(text) - 1, -1, -1):
   346→   346→        if text[i] in ('}', ']'):
   347→   347→            last_brace = i
   348→   348→            break
   349→   349→
   350→   350→    if last_brace >= 0 and last_brace < len(text) - 1:
   351→   351→        text = text[:last_brace + 1]
   352→   352→
   353→   353→    # إزالة الأسطر الجديدة الزائدة في البداية والنهاية
   354→   354→    text = text.strip()
   355→   355→
   356→   356→    return text

   926→   926→        """
   927→   927→        يفحص النص ويعيد:

   930→   930→        """

   958→   958→            matches = pattern.findall(text_str)
   959→   959→            found_keywords.extend(matches)
   960→   960→
   961→   961→        # إزالة التكرار
   962→   962→        found_keywords = list(dict.fromkeys(found_keywords))
   963→   963→
   964→   964→        # يجب أن يحتوي على كلمة واحدة على الأقل
