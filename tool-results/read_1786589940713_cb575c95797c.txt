    57→    57→function detectCountry(text: string): string | null {
    58→    58→  if (!text) return null
    59→    59→  const lower = text.toLowerCase()
    60→    60→  for (const [country, keywords] of Object.entries(COUNTRY_KEYWORDS)) {
    61→    61→    for (const kw of keywords) {
    62→    62→      if (lower.includes(kw.toLowerCase())) return country
    63→    63→    }
    64→    64→  }
    65→    65→  return null
    66→    66→}

    70→    70→ * React 19 blocks javascript: URLs by default, but explicit validation

    73→    73→function safeUrl(url: string | null | undefined): string | null {
    74→    74→  if (!url || typeof url !== 'string') return null
    75→    75→  const trimmed = url.trim()
    76→    76→  if (/^https?:\/\//i.test(trimmed)) return trimmed
    77→    77→  return null
    78→    78→}
