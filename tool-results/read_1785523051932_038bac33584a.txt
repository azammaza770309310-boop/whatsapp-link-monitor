   327→   926→        """
   328→   927→        يفحص النص ويعيد:
   329→   928→        - True + كلمات مطابقة إن كان طلب مساعدة
   330→   929→        - False + [] إن لم يكن
   331→   930→        """

   333→   932→            return False, []
   334→   933→
   335→   934→        text_str = text.strip()

   337→   936→            return False, []

   343→   942→                return False, []

   357→   956→        if single_keywords:

   359→   958→            matches = pattern.findall(text_str)
   360→   959→            found_keywords.extend(matches)
   361→   960→
   362→   961→        # إزالة التكرار
   363→   962→        found_keywords = list(dict.fromkeys(found_keywords))
   364→   963→
   365→   964→        # يجب أن يحتوي على كلمة واحدة على الأقل
   366→   965→        if len(found_keywords) >= 1:
