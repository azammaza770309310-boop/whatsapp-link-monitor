   669→class DatabaseManager:
   670→    def __init__(self, db_path=DB_FILE):
   671→        self.db_path = db_path
   672→        self._conn = None
   673→        self._lock = asyncio.Lock()
   674→        # إعدادات Supabase
   675→        self.supabase_url = os.getenv("SUPABASE_URL", "")
   676→        self.supabase_key = os.getenv("SUPABASE_KEY", "")
   677→        self._supabase_session = None

   679→    async def _get_supabase_session(self):
   680→        if self._supabase_session is None or self._supabase_session.closed:
   681→            self._supabase_session = aiohttp.ClientSession(headers={
   682→                "apikey": self.supabase_key,
   683→                "Authorization": f"Bearer {self.supabase_key}",
   684→                "Content-Type": "application/json",
   685→                "Prefer": "return=minimal"
   686→            })
   687→        return self._supabase_session
   688→
   689→    async def _supabase_insert_link(self, link, link_type, message_text, group_name,
   690→                                     sender_name, sender_contact, source_phone, message_link):
   691→        """إرسال الرابط إلى Supabase"""
   692→        if not self.supabase_url or not self.supabase_key:
   693→            return
   694→        try:
   695→            session = await self._get_supabase_session()
   696→            data = {
   697→                "link": link,
   698→                "link_type": link_type,
   699→                "message_text": message_text[:500] if message_text else None,
   700→                "group_name": group_name,
   701→                "sender_name": sender_name,
   702→                "sender_contact": sender_contact,
   703→                "source_phone": source_phone,
   704→                "message_link": message_link
   705→            }
   706→            async with session.post(f"{self.supabase_url}/rest/v1/links", json=data) as resp:
   707→                if resp.status not in (200, 201):
   708→                    text = await resp.text()
   709→                    if "duplicate" not in text.lower():
   710→                        logging.error(f"Supabase link insert: {resp.status} - {text[:100]}")
   711→        except Exception as e:
   712→            logging.error(f"Supabase insert exception: {e}")
   713→
   714→    async def _supabase_add_watcher(self, phone, display_name, session_string):
   715→        """إرسال المراقب إلى Supabase"""
   716→        if not self.supabase_url or not self.supabase_key:
   717→            return
   718→        try:
   719→            session = await self._get_supabase_session()
   720→            data = {
   721→                "phone": phone,
   722→                "display_name": display_name,
   723→                "session_string": session_string,
   724→                "is_active": True
   725→            }
   726→            # محاولة إدراج
   727→            async with session.post(f"{self.supabase_url}/rest/v1/watchers", json=data) as resp:
   728→                if resp.status not in (200, 201):
   729→                    text = await resp.text()
   730→                    if "duplicate" in text.lower():
   731→                        # تحديث الموجود (upsert semantics)
   732→                        async with session.patch(
   733→                            f"{self.supabase_url}/rest/v1/watchers?phone=eq.{phone}",
   734→                            json={"display_name": display_name, "session_string": session_string, "is_active": True}
   735→                        ):
   736→                            pass
   737→        except Exception as e:
   738→            logging.error(f"Supabase watcher exception: {e}")
   739→
   740→    async def _supabase_get_watchers(self):
   741→        """جلب المراقبين من Supabase (المصدر الأساسي)"""
   742→        if not self.supabase_url or not self.supabase_key:
   743→            return None  # استخدم المحلي
   744→        try:
   745→            session = await self._get_supabase_session()
   746→            async with session.get(
   747→                f"{self.supabase_url}/rest/v1/watchers?is_active=eq.true&select=phone,display_name,session_string"
   748→            ) as resp:
   749→                if resp.status == 200:
   750→                    data = await resp.json()
   751→                    if data:
   752→                        return data
   753→                    return []
   754→                return None
   755→        except Exception as e:
   756→            logging.error(f"Supabase get watchers: {e}")
   757→            return None

   759→    async def _supabase_count_links(self):
   760→        """عد الروابط من Supabase"""
   761→        if not self.supabase_url or not self.supabase_key:
   762→            return None
   763→        try:
   764→            session = await self._get_supabase_session()
   765→            headers = {**session.headers, "Prefer": "count=exact"}
   766→            async with session.get(
   767→                f"{self.supabase_url}/rest/v1/links?select=id&limit=1",
   768→                headers=headers
   769→            ) as resp:
   770→                range_header = resp.headers.get("content-range", "0/0")
   771→                return int(range_header.split("/")[-1] or 0)
   772→        except (aiohttp.ClientError, ValueError, TypeError) as e:
   773→            logging.error(f"Supabase count_links: {e}")
   774→            return None

   776→    async def _ensure_conn(self):
   777→        if self._conn is None:
   778→            Path(self.db_path).parent.mkdir(parents=True, exist_ok=True)

   780→            await self._conn.execute("PRAGMA journal_mode=WAL")
   781→            await self._conn.execute("PRAGMA busy_timeout=30000")
   782→            await self._conn.execute("PRAGMA synchronous=NORMAL")
   783→        return self._conn

   818→    async def add_watcher(self, phone: str, display_name: str, session_string: str) -> bool:
   819→        """إضافة مستخدم مراقب جديد - يحفظ في SQLite + Supabase"""

   823→        async with self._lock:
   824→            conn = await self._ensure_conn()
   825→            try:
   826→                cursor = await conn.execute("SELECT phone FROM watchers WHERE phone = ?", (phone,))
   827→                existing = await cursor.fetchone()
   828→                if existing:
   829→                    await conn.execute(
   830→                        """UPDATE watchers SET display_name = ?, session_string = ?, is_active = 1 WHERE phone = ?""",
   831→                        (display_name, session_string, phone))
   832→                else:
   833→                    await conn.execute(
   834→                        """INSERT INTO watchers (phone, display_name, session_string, is_active) VALUES (?, ?, ?, 1)""",
   835→                        (phone, display_name, session_string))
   836→                await conn.commit()
   837→                return True

   839→                logging.error(f"Add watcher error: {e}")

   861→    async def insert_request(self, link: str, message_date: datetime,
   862→                              group_name: str, sender_name: str, source_phone: str,
   863→                              message_link: str = None, message_text: str = None,
   864→                              sender_contact: str = None, link_type: str = None) -> bool:
   865→        """إدراج رابط جديد - يتحقق من التكرار أولاً ثم يحفظ"""
   866→        # تحديد نوع الرابط
   867→        if not link_type:
   868→            link_lower = link.lower()
   869→            if "chat.whatsapp.com" in link_lower or "wa.me" in link_lower or "whatsapp.com" in link_lower:
   870→                link_type = "whatsapp"
   871→            elif "t.me" in link_lower or "telegram.me" in link_lower:
   872→                link_type = "telegram"
   873→            else:
   874→                link_type = "other"
   875→
   876→        # 1. التحقق من التكرار محلياً أولاً (سريع جداً)
   877→        async with self._lock:
   878→            conn = await self._ensure_conn()
   879→            normalized_link = link.lower().strip().rstrip("/")
   880→            # usedforsecurity=False: this is a deduplication key, not a security primitive
   881→            content_hash = hashlib.md5(normalized_link.encode(), usedforsecurity=False).hexdigest()
   882→            
   883→            # فحص إذا كان الرابط موجوداً مسبقاً
   884→            cursor = await conn.execute(
   885→                "SELECT id FROM forwarded_requests WHERE content_hash = ?", (content_hash,))
   886→            existing = await cursor.fetchone()
   887→            if existing:
   888→                # مكرر - لا تحفظ ولا ترسل
   889→                return False
   890→
   891→        # 2. حفظ في Supabase (فقط بعد التأكد أنه غير مكرر)
   892→        await self._supabase_insert_link(
   893→            link, link_type, message_text, group_name,
   894→            sender_name, sender_contact, source_phone, message_link)
   895→
   896→        # 3. حفظ محلياً
   897→        async with self._lock:
   898→            try:
   899→                await conn.execute(
   900→                    """INSERT OR IGNORE INTO forwarded_requests
   901→                    (message_text, message_date, group_name, sender_name, source_phone, message_link, content_hash)
   902→                    VALUES (?, ?, ?, ?, ?, ?, ?)""",
   903→                    (link, message_date.isoformat() if message_date else None,
   904→                     group_name, sender_name, source_phone, message_link, content_hash))
   905→                await conn.commit()
   906→                cursor = await conn.execute("SELECT changes()")
   907→                changes = await cursor.fetchone()
   908→                return changes[0] > 0
   909→            except Exception as e:
   910→                logging.error(f"Insert request error: {e}")
   911→                return False
   912→
   913→    async def count_requests(self, source_phone: str = None) -> int:

   969→class HelpRequestDetector:
   970→    """يكشف ما إذا كانت الرسالة تحتوي على طلب مساعدة دراسية"""
   971→
   972→    @staticmethod
   973→    def is_help_request(text: str, min_length: int = 20, max_length: int = 2000) -> Tuple[bool, List[str]]:
   974→        """
   975→        يفحص النص ويعيد:
   976→        - True + كلمات مطابقة إن كان طلب مساعدة
   977→        - False + [] إن لم يكن
   978→        """
   979→        if not text:
   980→            return False, []
   981→
   982→        text_str = text.strip()
   983→        if len(text_str) < min_length or len(text_str) > max_length:
   984→            return False, []
   985→
   986→        # فحص السبام أولاً
   987→        text_lower = text_str.lower()
   988→        for spam in SPAM_KEYWORDS:
   989→            if spam.lower() in text_lower:
   990→                return False, []
   991→
   992→        # فحص الكلمات المفتاحية
   993→        found_keywords = []
   994→        text_lower = text_str.lower()
   995→
   996→        # فحص الكلمات المركبة أولاً
   997→        for kw in HELP_KEYWORDS:
   998→            if ' ' in kw:
   999→                if kw.lower() in text_lower:
  1000→                    found_keywords.append(kw)
  1001→
  1002→        # فحص الكلمات المنفردة عبر regex
  1003→        single_keywords = [kw for kw in HELP_KEYWORDS if ' ' not in kw]
  1004→        if single_keywords:
  1005→            pattern = re.compile(r'\b(' + '|'.join(re.escape(k) for k in single_keywords) + r')\b', re.IGNORECASE)
  1006→            matches = pattern.findall(text_str)
  1007→            found_keywords.extend(matches)
  1008→
  1009→        # إزالة التكرار
  1010→        found_keywords = list(dict.fromkeys(found_keywords))
  1011→
  1012→        # يجب أن يحتوي على كلمة واحدة على الأقل
  1013→        if len(found_keywords) >= 1:
  1014→            return True, found_keywords
  1015→        return False, []

  1028→        """تنسيق آمن باستخدام blockquote مع حماية من HTML injection"""
  1029→        # تنظيف all user input من HTML
  1030→        safe_group = html_module.escape(str(group_name or "غير معروف"))
  1031→        safe_sender = html_module.escape(str(sender_name or "غير معروف"))
  1032→        safe_contact = html_module.escape(str(sender_contact or ""))
  1033→        safe_source = html_module.escape(str(source_phone or ""))

  1035→        if len(message_text or "") > 150:
  1036→            safe_text += "..."
  1037→
  1038→        date_str = message_date.strftime("%Y-%m-%d %H:%M") if message_date else "غير معروف"
  1039→
  1040→        # تحديد نوع الرابط

  1046→        else:
  1047→            link_type_str = "🔗 رابط"
  1048→
  1049→        # URL validation: only allow http(s) and telegram/whatsapp schemes to prevent href injection
  1050→        def _safe_url(url: str) -> str:
  1051→            """Return URL only if it has an allowed scheme, else return empty string."""
  1052→            if not url:
  1053→                return ""
  1054→            url = url.strip()
  1055→            # Allow only http(s) URLs — telegram/whatsapp invite links are http(s)
  1056→            if url.lower().startswith(("http://", "https://")):
  1057→                # Escape quotes/angle brackets for use inside HTML attribute
  1058→                return html_module.escape(url, quote=True)
  1059→            return ""
  1060→
  1061→        safe_link = _safe_url(link)
  1062→        safe_msg_link = _safe_url(message_link) if message_link else ""
  1063→
  1064→        # بناء blockquote واحد شامل
  1065→        content = f"<b>🔗 رابط محفوظ ({link_type_str})</b>\n\n"
  1066→        content += f"👥 <b>العضوية:</b> {safe_group}\n"
