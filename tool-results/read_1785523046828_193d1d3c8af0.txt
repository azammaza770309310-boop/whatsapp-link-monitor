   815→    async def insert_request(self, link: str, message_date: datetime,
   816→                              group_name: str, sender_name: str, source_phone: str,
   817→                              message_link: str = None, message_text: str = None,
   818→                              sender_contact: str = None, link_type: str = None) -> bool:
   819→        """إدراج رابط جديد - يتحقق من التكرار أولاً ثم يحفظ"""
   820→        # تحديد نوع الرابط
   821→        if not link_type:
   822→            link_lower = link.lower()
   823→            if "chat.whatsapp.com" in link_lower or "wa.me" in link_lower or "whatsapp.com" in link_lower:
   824→                link_type = "whatsapp"
   825→            elif "t.me" in link_lower or "telegram.me" in link_lower:
   826→                link_type = "telegram"
   827→            else:
   828→                link_type = "other"
   829→
   830→        # 1. التحقق من التكرار محلياً أولاً (سريع جداً)
   831→        async with self._lock:
   832→            conn = await self._ensure_conn()
   833→            import hashlib
   834→            normalized_link = link.lower().strip().rstrip("/")
   835→            content_hash = hashlib.md5(normalized_link.encode()).hexdigest()
   836→            
   837→            # فحص إذا كان الرابط موجوداً مسبقاً
   838→            cursor = await conn.execute(
   839→                "SELECT id FROM forwarded_requests WHERE content_hash = ?", (content_hash,))
   840→            existing = await cursor.fetchone()
   841→            if existing:
   842→                # مكرر - لا تحفظ ولا ترسل
   843→                return False
   844→
   845→        # 2. حفظ في Supabase (فقط بعد التأكد أنه غير مكرر)
   846→        await self._supabase_insert_link(
   847→            link, link_type, message_text, group_name,
   848→            sender_name, sender_contact, source_phone, message_link)
   849→
   850→        # 3. حفظ محلياً
   851→        async with self._lock:
   852→            try:
   853→                await conn.execute(
   854→                    """INSERT OR IGNORE INTO forwarded_requests
   855→                    (message_text, message_date, group_name, sender_name, source_phone, message_link, content_hash)
   856→                    VALUES (?, ?, ?, ?, ?, ?, ?)""",
   857→                    (link, message_date.isoformat() if message_date else None,
   858→                     group_name, sender_name, source_phone, message_link, content_hash))
   859→                await conn.commit()
   860→                cursor = await conn.execute("SELECT changes()")
   861→                changes = await cursor.fetchone()
   862→                return changes[0] > 0
   863→            except Exception as e:
   864→                logging.error(f"Insert request error: {e}")
   865→                return False

   921→class HelpRequestDetector:
   922→    """يكشف ما إذا كانت الرسالة تحتوي على طلب مساعدة دراسية"""
   923→
   924→    @staticmethod
   925→    def is_help_request(text: str, min_length: int = 20, max_length: int = 2000) -> Tuple[bool, List[str]]:
   926→        """
   927→        يفحص النص ويعيد:
   928→        - True + كلمات مطابقة إن كان طلب مساعدة
   929→        - False + [] إن لم يكن
   930→        """
   931→        if not text:
   932→            return False, []
   933→
   934→        text_str = text.strip()
   935→        if len(text_str) < min_length or len(text_str) > max_length:
   936→            return False, []
   937→
   938→        # فحص السبام أولاً
   939→        text_lower = text_str.lower()
   940→        for spam in SPAM_KEYWORDS:
   941→            if spam.lower() in text_lower:
   942→                return False, []
   943→
   944→        # فحص الكلمات المفتاحية
   945→        found_keywords = []
   946→        text_lower = text_str.lower()
   947→
   948→        # فحص الكلمات المركبة أولاً
   949→        for kw in HELP_KEYWORDS:
   950→            if ' ' in kw:
   951→                if kw.lower() in text_lower:
   952→                    found_keywords.append(kw)
   953→
   954→        # فحص الكلمات المنفردة عبر regex
   955→        single_keywords = [kw for kw in HELP_KEYWORDS if ' ' not in kw]
   956→        if single_keywords:
   957→            pattern = re.compile(r'\b(' + '|'.join(re.escape(k) for k in single_keywords) + r')\b', re.IGNORECASE)
   958→            matches = pattern.findall(text_str)
   959→            found_keywords.extend(matches)
   960→
   961→        # إزالة التكرار
   962→        found_keywords = list(dict.fromkeys(found_keywords))
   963→
   964→        # يجب أن يحتوي على كلمة واحدة على الأقل
   965→        if len(found_keywords) >= 1:
   966→            return True, found_keywords
   967→        return False, []
