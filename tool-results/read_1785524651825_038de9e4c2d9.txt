   262→   861→    async def insert_request(self, link: str, message_date: datetime,
   263→   862→                              group_name: str, sender_name: str, source_phone: str,
   264→   863→                              message_link: str = None, message_text: str = None,
   265→   864→                              sender_contact: str = None, link_type: str = None) -> bool:
   266→   865→        """إدراج رابط جديد - يتحقق من التكرار أولاً ثم يحفظ"""
   267→   866→        # تحديد نوع الرابط
   268→   867→        if not link_type:
   269→   868→            link_lower = link.lower()
   270→   869→            if "chat.whatsapp.com" in link_lower or "wa.me" in link_lower or "whatsapp.com" in link_lower:
   271→   870→                link_type = "whatsapp"
   272→   871→            elif "t.me" in link_lower or "telegram.me" in link_lower:
   273→   872→                link_type = "telegram"
   274→   873→            else:
   275→   874→                link_type = "other"
   276→   875→
   277→   876→        # 1. التحقق من التكرار محلياً أولاً (سريع جداً)
   278→   877→        async with self._lock:
   279→   878→            conn = await self._ensure_conn()
   280→   879→            normalized_link = link.lower().strip().rstrip("/")
   281→   880→            # usedforsecurity=False: this is a deduplication key, not a security primitive
   282→   881→            content_hash = hashlib.md5(normalized_link.encode(), usedforsecurity=False).hexdigest()
   283→   882→            
   284→   883→            # فحص إذا كان الرابط موجوداً مسبقاً
   285→   884→            cursor = await conn.execute(
   286→   885→                "SELECT id FROM forwarded_requests WHERE content_hash = ?", (content_hash,))
   287→   886→            existing = await cursor.fetchone()
   288→   887→            if existing:
   289→   888→                # مكرر - لا تحفظ ولا ترسل
   290→   889→                return False
   291→   890→
   292→   891→        # 2. حفظ في Supabase (فقط بعد التأكد أنه غير مكرر)
   293→   892→        await self._supabase_insert_link(
   294→   893→            link, link_type, message_text, group_name,
   295→   894→            sender_name, sender_contact, source_phone, message_link)
   296→   895→
   297→   896→        # 3. حفظ محلياً
   298→   897→        async with self._lock:
   299→   898→            try:
   300→   899→                await conn.execute(
   301→   900→                    """INSERT OR IGNORE INTO forwarded_requests
   302→   901→                    (message_text, message_date, group_name, sender_name, source_phone, message_link, content_hash)
   303→   902→                    VALUES (?, ?, ?, ?, ?, ?, ?)""",
   304→   903→                    (link, message_date.isoformat() if message_date else None,
   305→   904→                     group_name, sender_name, source_phone, message_link, content_hash))
   306→   905→                await conn.commit()
   307→   906→                cursor = await conn.execute("SELECT changes()")
   308→   907→                changes = await cursor.fetchone()
   309→   908→                return changes[0] > 0
   310→   909→            except Exception as e:
   311→   910→                logging.error(f"Insert request error: {e}")
   312→   911→                return False

   370→   969→class HelpRequestDetector:
   371→   970→    """يكشف ما إذا كانت الرسالة تحتوي على طلب مساعدة دراسية"""
   372→   971→
   373→   972→    @staticmethod
   374→   973→    def is_help_request(text: str, min_length: int = 20, max_length: int = 2000) -> Tuple[bool, List[str]]:
   375→   974→        """
   376→   975→        يفحص النص ويعيد:
   377→   976→        - True + كلمات مطابقة إن كان طلب مساعدة
   378→   977→        - False + [] إن لم يكن
   379→   978→        """
   380→   979→        if not text:
   381→   980→            return False, []
   382→   981→
   383→   982→        text_str = text.strip()
   384→   983→        if len(text_str) < min_length or len(text_str) > max_length:
   385→   984→            return False, []
   386→   985→
   387→   986→        # فحص السبام أولاً
   388→   987→        text_lower = text_str.lower()
   389→   988→        for spam in SPAM_KEYWORDS:
   390→   989→            if spam.lower() in text_lower:
   391→   990→                return False, []
   392→   991→
   393→   992→        # فحص الكلمات المفتاحية
   394→   993→        found_keywords = []
   395→   994→        text_lower = text_str.lower()
   396→   995→
   397→   996→        # فحص الكلمات المركبة أولاً
   398→   997→        for kw in HELP_KEYWORDS:
   399→   998→            if ' ' in kw:
   400→   999→                if kw.lower() in text_lower:
   401→  1000→                    found_keywords.append(kw)
   402→  1001→
