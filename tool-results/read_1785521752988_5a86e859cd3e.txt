   161→def extract_whatsapp_telegram_links(text: str) -> list:
   162→    """يستخرج روابط واتساب وتيليجرام فقط (مجموعات/قروبات فقط)"""
   163→    if not text:
   164→        return []
   165→
   166→    links = []
   167→
   168→    # روابط واتساب (دعوات المجموعات فقط: chat.whatsapp.com)
   169→    for match in WHATSAPP_LINK_PATTERN.findall(text):
   170→        link = match.rstrip(".,;:!?)]}>\"'")
   171→        link_lower = link.lower()
   172→        # السماح فقط بدعوات مجموعات واتساب (chat.whatsapp.com)
   173→        # استبعاد: wa.me (دردشات مباشرة)، api.whatsapp.com (إرسال)، روابط مختصرة
   174→        if "chat.whatsapp.com" not in link_lower:
   175→            continue
   176→        if link and link not in links:
   177→            links.append(link)
   178→
   179→    # روابط تيليجرام (مجموعات/قنوات فقط، وليست روابط رسائل أو أرقام)
   180→    for match in TELEGRAM_LINK_PATTERN.findall(text):
   181→        link = match.rstrip(".,;:!?)]}>\"'")
   182→        link_lower = link.lower()
   183→        # استبعاد روابط الانضمام للمجموعات الخاصة (t.me/+xxx)
   184→        if "/+" in link or "joinchat" in link_lower:
   185→            continue
   186→        # استبعاد روابط الرسائل المباشرة (t.me/c/xxx) - هذه روابط رسائل خاصة
   187→        if "/c/" in link_lower:
   188→            continue
   189→        # استبعاد روابط الدردشة المباشرة (t.me/username?start=xxx)
   190→        if "?start=" in link_lower or "?text=" in link_lower:
   191→            continue
   192→        if link and link not in links:
   193→            links.append(link)
   194→
   195→    return links

   307→def _extract_clean_json(text: str) -> str:
   308→    """
   309→    يستخرج JSON نظيف من نص قد يحتوي على علامات Markdown أو نص إضافي.
   310→    يتعامل مع:
   311→    - ```json ... ```
   312→    - ``` ... ```
   313→    - نص قبل/بعد الـ JSON
   314→    - الأسطر الجديدة الزائدة
   315→    """
   316→    if not text:
   317→        return ""
   318→
   319→    text = text.strip()
   320→
   321→    # إزالة علامات Markdown code block
   322→    if "```" in text:
   323→        # استخراج المحتوى بين علامات ```
   324→        import re as _re
   325→        # pattern يطابق ```json أو ``` في البداية، و ``` في النهاية
   326→        match = _re.search(r'```(?:json)?\s*\n?(.*?)```', text, _re.DOTALL)
   327→        if match:
   328→            text = match.group(1).strip()
   329→        else:
   330→            # لو فيه ``` بس ما اكتمل النمط، نزيلها يدوياً
   331→            text = text.replace("```json", "").replace("```", "").strip()
   332→
   333→    # إزالة أي نص قبل أول { أو [
   334→    first_brace = -1
   335→    for i, char in enumerate(text):
   336→        if char in ('{', '['):
   337→            first_brace = i
   338→            break
   339→
   340→    if first_brace > 0:
   341→        text = text[first_brace:]
   342→
   343→    # إزالة أي نص بعد آخر } أو ]
   344→    last_brace = -1
   345→    for i in range(len(text) - 1, -1, -1):
   346→        if text[i] in ('}', ']'):
   347→            last_brace = i
   348→            break
   349→
   350→    if last_brace >= 0 and last_brace < len(text) - 1:
   351→        text = text[:last_brace + 1]
   352→
   353→    # إزالة الأسطر الجديدة الزائدة في البداية والنهاية
   354→    text = text.strip()
   355→
   356→    return text

   412→    "sender_contact": "رقم الهاتف أو اليوزر أو فارغ",
   413→    "is_advertisement": true/false,
   414→    "country": "السعودية" أو "الكويت" أو "قطر" أو "البحرين" أو "الإمارات" أو "أخرى",
   415→    "description": "وصف مختصر في 5 كلمات"
   416→}}
   417→
   418→القواعد الصارمة:
   419→- should_save = true فقط إذا كان الرابط لمجموعة أو قناة تعليمية جامعية
   420→- should_save = false إذا كان الرابط غير تعليمي (تسوق، ترفيه، أخبار، إلخ)
   421→- should_save = false إذا كان دردشة مباشرة (wa.me/رقم بدون /message)
   422→- is_advertisement = true إذا كانت الرسالة ترويج لخدمات مدفوعة أو إعلان
   423→- is_advertisement = true إذا كان الرابط لخدمة مدفوعة (مكتب، مركز، شركة)
   424→- sender_contact = رقم الهاتف أو @اليوزر المذكور في الرسالة
   425→- استخرج الرابط الكامل بشكل صحيح من أي صيغة
   426→- فحص كل أنواع الروابط: chat.whatsapp.com, wa.me, t.me, telegram.me
   427→- إذا لم يوجد رابط واتساب أو تيليجرام، should_save = false
   428→- country: حدد الدولة بدقة من سياق الرسالة"""
   429→
   430→        # محاولة مع كل المفاتيح
   431→        for attempt in range(len(self.providers)):
   432→            provider = self.providers[self._current_provider]
   433→            try:
   434→                session = await self._get_session()
   435→                headers = {
   436→                    "Authorization": f"Bearer {provider['key']}",
   437→                    "Content-Type": "application/json"
   438→                }

   440→                    "model": provider["model"],
   441→                    "messages": [
   442→                        {"role": "system", "content": "أنت محلل رسائل ذكي. أعد JSON فقط."},

   444→                    ],
   445→                    "temperature": 0.1,
   446→                    "max_tokens": 300
   447→                }
   448→
   449→                async with session.post(provider["url"], json=payload, headers=headers,
   450→                                       timeout=aiohttp.ClientTimeout(total=15)) as resp:
   451→                    if resp.status == 200:
   452→                        data = await resp.json()
   453→                        content = data.get("choices", [{}])[0].get("message", {}).get("content", "")
   454→                        # تنظيف عميق للـ JSON من علامات Markdown
   455→                        clean_json = _extract_clean_json(content)
   456→                        result = json_module.loads(clean_json)
   457→                        logging.info(f"🤖 AI [{provider['name']}]: {result.get('should_save')} | {result.get('link_type')}")
   458→                        return result
   459→                    elif resp.status in (429, 401, 403, 404):
   460→                        # تبديل المفتاح
   461→                        logging.warning(f"AI [{provider['name']}] error {resp.status} - switching key")
   462→                        self._current_provider = (self._current_provider + 1) % len(self.providers)
   463→                        continue
   464→                    else:
   465→                        logging.warning(f"AI [{provider['name']}] error: {resp.status}")

   469→            except asyncio.TimeoutError:
   470→                logging.warning(f"AI [{provider['name']}] timeout - switching")
   471→                self._current_provider = (self._current_provider + 1) % len(self.providers)
   472→                continue
   473→            except Exception as e:
   474→                logging.error(f"AI [{provider['name']}] error: {e}")
   475→                self._current_provider = (self._current_provider + 1) % len(self.providers)
   476→                continue
   477→
   478→        # كل المفاتيح فشلت - استخدام الفلتر العادي
   479→        logging.warning("⚠️ All AI keys failed - using fallback")
   480→        return self._fallback_analysis(text)

   482→    @staticmethod
   483→    def _fallback_analysis(text: str) -> dict:
   484→        """تحليل بديل بدون AI - فحص شامل مثل الإنسان"""
   485→        if not text:
   486→            return {"should_save": False, "link": "", "link_type": "other",
   487→                    "sender_contact": "", "is_advertisement": False,
   488→                    "country": "أخرى", "description": ""}
   489→
   490→        # استخراج كل أنواع الروابط بكل الصيغ
   491→        links = []
   492→
   493→        # واتساب بكل الصيغ
   494→        wa_patterns = [
   495→            r'(?:https?://)?chat\.whatsapp\.com/[^\s<>"\)\]\']+',
   496→            r'(?:https?://)?wa\.me/[^\s<>"\)\]\']+',
   497→            r'(?:https?://)?whatsapp\.com/channel/[^\s<>"\)\]\']+',
   498→            r'(?:https?://)?whatsapp\.com/contact/[^\s<>"\)\]\']+',
   499→            r'(?:https?://)?api\.whatsapp\.com/[^\s<>"\)\]\']+',
   500→            r'(?:https?://)?l\.whatsapp\.com/[^\s<>"\)\]\']+',
   501→        ]
   502→        # تيليجرام بكل الصيغ
   503→        tg_patterns = [
   504→            r'(?:https?://)?t\.me/[^\s<>"\)\]\']+',
   505→            r'(?:https?://)?telegram\.me/[^\s<>"\)\]\']+',
   506→        ]
   507→
   508→        for pattern in wa_patterns + tg_patterns:
   509→            matches = re.findall(pattern, text, re.IGNORECASE)
   510→            for m in matches:
   511→                m = m.rstrip(".,;:!?)]}>\"'")
   512→                if m and m not in links:
   513→                    links.append(m)
   514→
   515→        if not links:
   516→            return {"should_save": False, "link": "", "link_type": "other",
   517→                    "sender_contact": "", "is_advertisement": False,
   518→                    "country": "أخرى", "description": ""}
   519→
   520→        is_ad = is_advertiser_message(text)
   521→        contact = extract_sender_contact(text)
   522→
   523→        link = links[0]
   524→        link_lower = link.lower()
   525→        if any(x in link_lower for x in ["whatsapp.com", "wa.me"]):
   526→            link_type = "whatsapp"
   527→        elif any(x in link_lower for x in ["t.me", "telegram.me"]):
   528→            link_type = "telegram"
   529→        else:
   530→            link_type = "other"
   531→
   532→        return {
   533→            "should_save": not is_ad,
   534→            "link": link,
   535→            "link_type": link_type,
   536→            "sender_contact": contact,
   537→            "is_advertisement": is_ad,
   538→            "country": "أخرى",
   539→            "description": ""
   540→        }

   806→    async def insert_request(self, link: str, message_date: datetime,
   807→                              group_name: str, sender_name: str, source_phone: str,
   808→                              message_link: str = None, message_text: str = None,
   809→                              sender_contact: str = None, link_type: str = None) -> bool:
   810→        """إدراج رابط جديد - يتحقق من التكرار أولاً ثم يحفظ"""
   811→        # تحديد نوع الرابط
   812→        if not link_type:
   813→            link_lower = link.lower()
   814→            if "chat.whatsapp.com" in link_lower or "wa.me" in link_lower or "whatsapp.com" in link_lower:
   815→                link_type = "whatsapp"
   816→            elif "t.me" in link_lower or "telegram.me" in link_lower:
   817→                link_type = "telegram"
   818→            else:
   819→                link_type = "other"
   820→
   821→        # 1. التحقق من التكرار محلياً أولاً (سريع جداً)
   822→        async with self._lock:
   823→            conn = await self._ensure_conn()
   824→            import hashlib
   825→            normalized_link = link.lower().strip().rstrip("/")
   826→            content_hash = hashlib.md5(normalized_link.encode()).hexdigest()
   827→            
   828→            # فحص إذا كان الرابط موجوداً مسبقاً
   829→            cursor = await conn.execute(
   830→                "SELECT id FROM forwarded_requests WHERE content_hash = ?", (content_hash,))
   831→            existing = await cursor.fetchone()
   832→            if existing:
   833→                # مكرر - لا تحفظ ولا ترسل
   834→                return False
   835→
   836→        # 2. حفظ في Supabase (فقط بعد التأكد أنه غير مكرر)
   837→        await self._supabase_insert_link(
   838→            link, link_type, message_text, group_name,
   839→            sender_name, sender_contact, source_phone, message_link)
   840→
   841→        # 3. حفظ محلياً
   842→        async with self._lock:
   843→            try:
   844→                await conn.execute(
   845→                    """INSERT OR IGNORE INTO forwarded_requests
   846→                    (message_text, message_date, group_name, sender_name, source_phone, message_link, content_hash)
   847→                    VALUES (?, ?, ?, ?, ?, ?, ?)""",
   848→                    (link, message_date.isoformat() if message_date else None,
   849→                     group_name, sender_name, source_phone, message_link, content_hash))
   850→                await conn.commit()
   851→                cursor = await conn.execute("SELECT changes()")
   852→                changes = await cursor.fetchone()
   853→                return changes[0] > 0
   854→            except Exception as e:
   855→                logging.error(f"Insert request error: {e}")
   856→                return False

   912→class HelpRequestDetector:
   913→    """يكشف ما إذا كانت الرسالة تحتوي على طلب مساعدة دراسية"""
   914→
   915→    @staticmethod
   916→    def is_help_request(text: str, min_length: int = 20, max_length: int = 2000) -> Tuple[bool, List[str]]:
   917→        """
   918→        يفحص النص ويعيد:
   919→        - True + كلمات مطابقة إن كان طلب مساعدة
   920→        - False + [] إن لم يكن
   921→        """
   922→        if not text:
   923→            return False, []
   924→
   925→        text_str = text.strip()
   926→        if len(text_str) < min_length or len(text_str) > max_length:
   927→            return False, []
   928→
   929→        # فحص السبام أولاً
   930→        text_lower = text_str.lower()
   931→        for spam in SPAM_KEYWORDS:
   932→            if spam.lower() in text_lower:
   933→                return False, []
   934→
   935→        # فحص الكلمات المفتاحية
   936→        found_keywords = []
   937→        text_lower = text_str.lower()
   938→
   939→        # فحص الكلمات المركبة أولاً
   940→        for kw in HELP_KEYWORDS:
   941→            if ' ' in kw:
   942→                if kw.lower() in text_lower:
   943→                    found_keywords.append(kw)
   944→
   945→        # فحص الكلمات المنفردة عبر regex
   946→        single_keywords = [kw for kw in HELP_KEYWORDS if ' ' not in kw]
   947→        if single_keywords:
   948→            pattern = re.compile(r'\b(' + '|'.join(re.escape(k) for k in single_keywords) + r')\b', re.IGNORECASE)
   949→            matches = pattern.findall(text_str)
   950→            found_keywords.extend(matches)
   951→
   952→        # إزالة التكرار
   953→        found_keywords = list(dict.fromkeys(found_keywords))
   954→
   955→        # يجب أن يحتوي على كلمة واحدة على الأقل
   956→        if len(found_keywords) >= 1:
   957→            return True, found_keywords
   958→        return False, []

  1149→    async def scan(self):
  1150→        start = datetime.now()
  1151→        if self.days_back is not None:
  1152→            hard = datetime.now() - timedelta(days=self.days_back)
  1153→        else: hard = None
  1154→        soft = None
  1155→        try:
  1156→            conn = await self.db._ensure_conn()
  1157→            cursor = await conn.execute(
  1158→                "SELECT MAX(last_scanned_message_date) FROM scan_state WHERE source_phone = ?",
  1159→                (self.source_phone,))
  1160→            row = await cursor.fetchone()
  1161→            if row and row[0]:
  1162→                soft = datetime.fromisoformat(row[0])
  1163→        except: pass
  1164→        eff = max(hard, soft) if (hard and soft) else (hard or soft)
  1165→        period = f"آخر {(datetime.now()-eff).days} يوم (متزايد)" if eff else "كامل"
  1166→        logging.info(f"[SCAN {self.source_phone}] Period: {period}")
  1167→
  1168→        try:
  1169→            dialogs = await self.user_client.get_dialogs()
  1170→        except Exception as e:
  1171→            logging.error(f"[SCAN {self.source_phone}] get_dialogs: {e}")
  1172→            return period
  1173→
  1174→        for idx, d in enumerate(dialogs, 1):
  1175→            if self._is_cancelled(): break
  1176→            if d.id == self.channel_id: continue
  1177→            if self.skip_channel_posts:
  1178→                try:
  1179→                    if d.is_channel: continue
  1180→                except: pass
  1181→            name = d.name or "Unknown"
  1182→            if self.progress_callback:
  1183→                try: self.progress_callback(idx, len(dialogs), name)
  1184→                except: pass
  1185→            try: await self._scan_chat(d, eff, name)
  1186→            except FloodWaitError as e:
  1187→                await asyncio.sleep(e.seconds + 1)
  1188→            except Exception as e:
  1189→                logging.error(f"[SCAN {self.source_phone}] Error {name}: {e}")
  1190→            await asyncio.sleep(0.3)
  1191→        dur = (datetime.now()-start).total_seconds()
  1192→        await self._send_summary(period, dur)
  1193→        return period

  1195→    async def _scan_chat(self, dialog, cutoff, name):

  1200→            try:
  1201→                ls = await self.db.get_last_scan_date(self.source_phone, dialog.id)

  1205→            async for msg in self.user_client.iter_messages(dialog, reverse=False, limit=self.max_per_chat):
  1206→                if self._is_cancelled(): break
  1207→                try: md = msg.date.replace(tzinfo=None) if msg.date else None
  1208→                except: md = None
  1209→                if md and chat_cut and md < chat_cut: break
  1210→                self.total_scanned += 1
  1211→                if md and (last_date is None or md > last_date): last_date = md
  1212→                if not msg or not msg.text: continue
  1213→
  1214→                # استخراج روابط واتساب وتيليجرام
  1215→                links = extract_whatsapp_telegram_links(msg.text)

  1217→

  1223→                if is_advertiser_message(msg.text):
  1224→                    continue

  1227→                try:
  1228→                    sender = await msg.get_sender()
  1229→                    sn = Monitor._get_sender_name(sender)

  1231→

  1235→                    contact = f"✈️ @{sender.username}"
  1236→
  1237→                # رابط الرسالة
  1238→                msg_link = None
  1239→                try:
  1240→                    msg_link = f"https://t.me/c/{str(dialog.id).replace('-100', '')}/{msg.id}"
  1241→                except: pass

  1246→                        link, md, name, sn, self.source_phone, msg_link,
  1247→                        message_text=msg.text, sender_contact=contact)
  1248→                    if inserted:
  1249→                        self.new_count += 1
  1250→                        batch.append({
  1251→                            'link': link, 'text': msg.text, 'date': md,
  1252→                            'group': name, 'sender': sn, 'msg_link': msg_link,
  1253→                            'contact': contact
  1254→                        })
  1255→                        if len(batch) >= self.batch_size:
  1256→                            await self._send_batch(batch)
  1257→                            batch = []
  1258→        except FloodWaitError: raise
  1259→        except Exception as e:
  1260→            logging.error(f"[SCAN {self.source_phone}] iter error: {e}")
  1261→            if last_date:
  1262→                try: await self.db.update_scan_state(self.source_phone, dialog.id, name, last_date)

  1267→            try: await self.db.update_scan_state(self.source_phone, dialog.id, name, last_date)
  1268→            except: pass
  1269→        self.chats_scanned += 1

  1506→    async def _check_telegram_membership(self, link: str) -> dict:
  1507→        """
  1508→        يفحص العضوية بدقة عالية وسرعة.
  1509→        يعيد: {phone: True(مشترك) / False(غير مشترك) / None(تعذر الفحص)}
  1510→        """
  1511→        results = {}
  1512→        
  1513→        # 1. استخراج username وتنظيفه
  1514→        link_lower = link.lower()
  1515→        if "t.me/+" in link or "joinchat" in link_lower:
  1516→            return {}  # روابط خاصة لا يمكن فحصها
  1517→        
  1518→        username = None
  1519→        if "t.me/" in link_lower:
  1520→            parts = link.split("t.me/", 1)
  1521→            if len(parts) > 1:
  1522→                username = parts[1].split("/")[0].split("?")[0].strip()
  1523→        
  1524→        if not username or len(username) < 5:
  1525→            return {}  # غالباً رابط مستخدم أو غير صالح
  1526→        
  1527→        # 2. بناء قاموس (Cache) لكل المراقبين: {username_lowercase: phone}
  1528→        # هذا أسرع بكثير من البحث في كل مرة
  1529→        watchers_dialogs = {}
  1530→        for phone, client in self.user_clients.items():
  1531→            if not client or not client.is_connected():
  1532→                results[phone] = None
  1533→                continue
  1534→            
  1535→            try:
  1536→                # جلب كل المجموعات/القنوات التي يشترك فيها هذا المراقب دفعة واحدة
  1537→                async for dialog in client.iter_dialogs(limit=500):
  1538→                    if hasattr(dialog.entity, 'username') and dialog.entity.username:
  1539→                        watchers_dialogs[dialog.entity.username.lower()] = phone
  1540→                    # أيضاً التحقق من المعرف الرقمي (للمجموعات بدون يوزرنيم)
  1541→                    if hasattr(dialog.entity, 'id'):
  1542→                        watchers_dialogs[f"id_{dialog.entity.id}"] = phone
  1543→            except Exception as e:
  1544→                logging.warning(f"[MEMBERSHIP] Failed to fetch dialogs for {phone}: {e}")
  1545→                results[phone] = None
  1546→        
  1547→        # 3. الآن نفحص الرابط المحدد
  1548→        username_lower = username.lower()
  1549→        
  1550→        for phone, client in self.user_clients.items():
  1551→            if phone in results and results[phone] is None:
  1552→                continue  # تم تعليمه كغير متصل
  1553→                
  1554→            try:
  1555→                # محاولة الفحص المباشر عبر API (الأدق)
  1556→                try:
  1557→                    from telethon.tl.functions.channels import GetParticipantRequest
  1558→                    entity = await client.get_entity(username)
  1559→                    
  1560→                    # التأكد أنه ليس مستخدم عادي
  1561→                    if hasattr(entity, 'first_name') and not hasattr(entity, 'megagroup') and not hasattr(entity, 'broadcast'):
  1562→                        results[phone] = True  # مستخدم، اعتبره مشترك لتجاهله
  1563→                        continue
  1564→                    
  1565→                    try:
  1566→                        await client(GetParticipantRequest(channel=entity, participant="me"))
  1567→                        results[phone] = True  # مشترك تأكيداً
  1568→                    except Exception:
  1569→                        results[phone] = False  # تأكد عدم الاشتراك
  1570→                except Exception:
  1571→                    # لو فشل الـ API، نعتمد على الـ Cache الذي بنيناه
  1572→                    if username_lower in watchers_dialogs and watchers_dialogs[username_lower] == phone:
  1573→                        results[phone] = True
  1574→                    else:
  1575→                        results[phone] = False
  1576→                        
  1577→            except Exception as e:
  1578→                logging.error(f"[MEMBERSHIP] Error checking {phone}: {e}")
  1579→                results[phone] = None
  1580→        
  1581→        logging.info(f"[MEMBERSHIP] Check for @{username}: {results}")
  1582→        return results

  1705→    async def _send(self, text, retries=3, buttons=None, parse_mode='html'):
  1706→        """إرسال رسالة للقناة مع دعم HTML والأزرار"""
  1707→        async with self._send_lock:
  1708→            for a in range(1, retries + 1):
  1709→                try:
  1710→                    await self.bot_client.send_message(
  1711→                        self.config.channel_id, text,
  1712→                        parse_mode=parse_mode,
  1713→                        buttons=buttons,
  1714→                        link_preview=False
  1715→                    )
  1716→                    return
  1717→                except FloodWaitError as e:
  1718→                    await asyncio.sleep(e.seconds + 1)
  1719→                except (RPCError, OSError, ConnectionError):
  1720→                    await asyncio.sleep(min(10 * a, 60))
  1721→            logging.error(f"Failed after {retries} attempts")
