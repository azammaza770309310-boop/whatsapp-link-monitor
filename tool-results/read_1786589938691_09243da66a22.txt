    57→function detectCountry(text: string): string | null {
    58→  if (!text) return null
    59→  const lower = text.toLowerCase()
    60→  for (const [country, keywords] of Object.entries(COUNTRY_KEYWORDS)) {
    61→    for (const kw of keywords) {
    62→      if (lower.includes(kw.toLowerCase())) return country
    63→    }
    64→  }
    65→  return null
    66→}

    68→/**
    69→ * Defense-in-depth: only allow http(s) URLs in href attributes.
    70→ * React 19 blocks javascript: URLs by default, but explicit validation
    71→ * prevents any future regression and handles edge cases (data: URIs, etc).
    72→ */
    73→function safeUrl(url: string | null | undefined): string | null {
    74→  if (!url || typeof url !== 'string') return null
    75→  const trimmed = url.trim()
    76→  if (/^https?:\/\//i.test(trimmed)) return trimmed
    77→  return null
    78→}

   545→    // استخدم API endpoint أولاً (المصدر الحقيقي من SQLite group_states)

   547→      const response = await fetch(`${API_URL}/api/joined_groups`, {
   548→        headers: { 'Accept': 'application/json' }
   549→      })
   550→      if (response.ok) {
   551→        const data = await response.json()
   552→        const groups = data.joined_groups || []
   553→        const stats = data.stats || {}
   554→        setJoinedGroups(groups)
   555→        setJoinerStats({
   556→          total_joined: stats.total_joined || 0,
   557→          pending_groups: stats.pending_groups || 0,
   558→          active_joiners: stats.active_joiners || 0
   559→        })
   560→        setLoading(false)
   561→        return
   562→      }
   563→    } catch {
   564→      // API غير متاح، جرب Supabase
   565→    }
   566→    // Fallback: Supabase target_groups
